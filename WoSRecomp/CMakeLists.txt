# The host harness — see main.cpp.
#
# Standalone project, mirroring tools/xex_info: it pulls in XenonUtils (for XEX
# parsing, so the harness loads the image with exactly the same code the
# recompiler used) and WoSRecompLib (the recompiled game code).
#
# NOTE: this links cleanly with nothing implemented, and that is NOT a good
# sign — it used to say the opposite here. Every recompiled function is
# emitted as a weak alias, so no game function is ever undefined, and
# XenonUtils rewrites each import thunk to nop/nop/nop/blr, so unimplemented
# imports return silently rather than failing to link. The runtime to-do list
# comes from the XEX import table (xex_info --imports) and from the boot-order
# trace the harness prints, not from linker errors. See main.cpp.
#
# The one thing the link DOES catch is inconsistency in the generated code —
# a duplicate symbol here means WoSRecompLib/ppc/ holds chunk files from two
# different recompiler runs. Re-run tools/recompile.sh, which prunes them.

cmake_minimum_required(VERSION 3.20)
project("WoSRecomp" CXX C)

set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)

set(XENONRECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../tools/XenonRecomp")
set(THIRDPARTY_ROOT "${XENONRECOMP_ROOT}/thirdparty")

add_subdirectory("${THIRDPARTY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/thirdparty")
add_subdirectory("${XENONRECOMP_ROOT}/XenonUtils" "${CMAKE_CURRENT_BINARY_DIR}/XenonUtils")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../WoSRecompLib" "${CMAKE_CURRENT_BINARY_DIR}/WoSRecompLib")

if(NOT TARGET WoSRecompLib)
    message(FATAL_ERROR
        "WoSRecompLib produced no target — the recompiled sources in "
        "WoSRecompLib/ppc/ are missing. Run XenonRecomp first.")
endif()

set(WOS_SOURCES
    main.cpp
    kernel/import_log.cpp
    # Real kernel implementations. Each function implemented here must also be
    # listed in kernel/kernel_overrides.h, which compiles out the matching
    # generated stub — otherwise the two collide at link time.
    kernel/panic.cpp
    kernel/format.cpp
    kernel/memory.cpp
    kernel/pcr.cpp
    kernel/system.cpp
    kernel/object.cpp
    kernel/sync.cpp
    kernel/thread.cpp
    kernel/file.cpp
    kernel/video.cpp
    kernel/debug.cpp
    kernel/rtl.cpp
    kernel/audio.cpp
    kernel/d3d_probe.cpp
    # Reads the guest's own globals rather than reporting what the host did.
    kernel/state_probe.cpp
    # Overrides *guest* functions rather than imports, to observe that control
    # reached a particular address. See the comment at the top of the file.
    kernel/trace_guest.cpp)

# This list is deliberately explicit rather than a GLOB: a file that silently
# joins the build is a file nobody reviewed. The cost is that adding a source
# and forgetting this line gives an "undefined symbol" at link — which has now
# happened once, so: if you added a .cpp under kernel/, add it here too.

# Generated by: xex_info <xex> --emit-stubs WoSRecomp/kernel/imports_generated.cpp
# Absent on a fresh clone; the host still builds without it, the game just has
# no import stubs and therefore no trace.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/kernel/imports_generated.cpp")
    # A file generated before the override mechanism existed has unguarded
    # stubs, which collide with everything in kernel_overrides.h and produce a
    # wall of duplicate-symbol errors that says nothing about the real cause.
    # Catch it here with a message that does.
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/kernel/imports_generated.cpp" _wos_stub_text)
    string(FIND "${_wos_stub_text}" "WOS_IMPL_" _wos_guard_pos)
    if(_wos_guard_pos EQUAL -1)
        message(FATAL_ERROR
            "WoSRecomp/kernel/imports_generated.cpp predates the import override "
            "mechanism — it has no #ifndef WOS_IMPL_ guards, so its stubs would "
            "collide with the implementations in kernel/. Regenerate it:\n"
            "  ./tools/build_tools.sh\n"
            "  tools/xex_info/build/xex_info private/default.xex --emit-stubs "
            "WoSRecomp/kernel/imports_generated.cpp")
    endif()
    unset(_wos_stub_text)

    list(APPEND WOS_SOURCES kernel/imports_generated.cpp)
    message(STATUS "WoSRecomp: import stubs present")
else()
    message(WARNING
        "WoSRecomp: kernel/imports_generated.cpp missing — no import tracing. "
        "Generate it with: xex_info private/default.xex --emit-stubs "
        "WoSRecomp/kernel/imports_generated.cpp")
    list(APPEND WOS_SOURCES kernel/import_stub_count_fallback.cpp)
endif()

add_executable(WoSRecomp ${WOS_SOURCES})
target_include_directories(WoSRecomp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/kernel")

target_compile_definitions(WoSRecomp PRIVATE NOMINMAX)
target_link_libraries(WoSRecomp PRIVATE WoSRecompLib XenonUtils)

if(WIN32)
    # dbghelp: symbolises the host stack so a runaway guest stack reports the
    # actual sub_XXXXXXXX cycle instead of "it recursed somewhere".
    target_link_libraries(WoSRecomp PRIVATE dbghelp)

    # Recompiled guest functions are ordinary C++ calls, so guest call depth is
    # host stack depth. The default 1 MiB overflows after a few thousand guest
    # frames — and a stack overflow leaves no room to run an exception filter,
    # so the process dies silently and takes the whole trace with it. 256 MiB
    # of reserved (not committed) stack lets the runaway detector in main.cpp
    # win the race and actually report.
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_link_options(WoSRecomp PRIVATE /STACK:268435456)
    else()
        target_link_options(WoSRecomp PRIVATE -Wl,--stack,268435456)
    endif()
endif()

if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(WoSRecomp PRIVATE /clang:-march=sandybridge)
else()
    target_compile_options(WoSRecomp PRIVATE -march=sandybridge)
endif()
